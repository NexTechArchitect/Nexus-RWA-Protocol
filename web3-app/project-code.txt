"use client";

import { useEffect, useRef } from "react";
import type { MutableRefObject } from "react";
import * as THREE from "three";
import { createParticleEngine } from "../particles/ParticleEngine";
import { createFeatureScene } from "../scenes/FeatureScene";
import { createFinalScene } from "../scenes/FinalScene";
import { createHeroScene } from "../scenes/HeroScene";
import { createEnvironment, type ExperienceFrame, type SceneController } from "../world/Environment";

interface ExperienceCanvasProps {
  progressRef: MutableRefObject<number>;
  reducedMotion: boolean;
}

const clamp = (value: number) => Math.min(Math.max(value, 0), 1);
const damp = (current: number, target: number, speed: number, delta: number) =>
  THREE.MathUtils.lerp(current, target, 1 - Math.exp(-speed * delta));

function createCameraRails() {
  return {
    positions: new THREE.CatmullRomCurve3(
      [
        new THREE.Vector3(0.2, 0.3, 10.8),
        new THREE.Vector3(-1.5, 0.8, 8.2),
        new THREE.Vector3(2.0, 0.32, 5.25),
        new THREE.Vector3(5.1, 1.15, 6.4),
        new THREE.Vector3(-2.55, 2.15, 8.5),
        new THREE.Vector3(0.2, 0.48, 10.9),
      ],
      false,
      "catmullrom",
      0.42,
    ),
    targets: new THREE.CatmullRomCurve3(
      [
        new THREE.Vector3(0, 0.08, 0),
        new THREE.Vector3(0.1, 0.16, -0.15),
        new THREE.Vector3(0.25, -0.1, 0.15),
        new THREE.Vector3(0.65, 0.08, -0.4),
        new THREE.Vector3(-0.2, 0.42, 0.3),
        new THREE.Vector3(0, 0.02, 0.25),
      ],
      false,
      "catmullrom",
      0.4,
    ),
  };
}

export function ExperienceCanvas({ progressRef, reducedMotion }: ExperienceCanvasProps) {
  const hostRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const host = hostRef.current;

    if (!host) {
      return;
    }

    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(39, 1, 0.1, 90);
    const renderer = new THREE.WebGLRenderer({
      antialias: true,
      alpha: false,
      powerPreference: "high-performance",
    });

    renderer.outputColorSpace = THREE.SRGBColorSpace;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.08;
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 1.8));
    renderer.setClearColor("#f4f8f7", 1);
    host.appendChild(renderer.domElement);

    const world = createEnvironment(scene);
    const particles = createParticleEngine(scene);
    const controllers: SceneController[] = [
      createHeroScene(world),
      createFeatureScene(world),
      createFinalScene(world),
    ];

    const rails = createCameraRails();
    const targetPosition = new THREE.Vector3();
    const targetLookAt = new THREE.Vector3();
    const currentLookAt = new THREE.Vector3();
    const pointer = new THREE.Vector2();
    const targetPointer = new THREE.Vector2();
    const clock = new THREE.Clock();

    let animationFrame = 0;
    let pointerVelocity = 0;
    let previousPointer = new THREE.Vector2();

    const resize = () => {
      const bounds = host.getBoundingClientRect();
      renderer.setSize(Math.max(bounds.width, 1), Math.max(bounds.height, 1), false);
      camera.aspect = Math.max(bounds.width, 1) / Math.max(bounds.height, 1);
      camera.fov = bounds.width < 700 ? 46 : 39;
      camera.updateProjectionMatrix();
    };

    const move = (event: PointerEvent) => {
      const bounds = host.getBoundingClientRect();
      const next = new THREE.Vector2(
        ((event.clientX - bounds.left) / bounds.width - 0.5) * 2,
        -((event.clientY - bounds.top) / bounds.height - 0.5) * 2,
      );

      pointerVelocity = Math.min(next.distanceTo(previousPointer) * 2.4, 1);
      previousPointer = next;
      targetPointer.copy(next);
    };

    const resetPointer = () => targetPointer.set(0, 0);

    const render = () => {
      const delta = Math.min(clock.getDelta(), 0.05);
      const time = clock.elapsedTime;
      const progress = clamp(progressRef.current);

      pointer.lerp(targetPointer, 1 - Math.exp(-delta * 5.6));
      pointerVelocity = damp(pointerVelocity, 0, 4.5, delta);

      rails.positions.getPointAt(progress, targetPosition);
      rails.targets.getPointAt(progress, targetLookAt);

      targetPosition.x += pointer.x * 0.17 + (window.innerWidth < 700 ? 0.75 : 0);
      targetPosition.y += pointer.y * 0.11;
      targetLookAt.x += pointer.x * 0.13;
      targetLookAt.y += pointer.y * 0.09;

      camera.position.set(
        damp(camera.position.x, targetPosition.x, reducedMotion ? 3.2 : 1.95, delta),
        damp(camera.position.y, targetPosition.y, reducedMotion ? 3.2 : 1.95, delta),
        damp(camera.position.z, targetPosition.z, reducedMotion ? 3.2 : 1.95, delta),
      );

      currentLookAt.set(
        damp(currentLookAt.x, targetLookAt.x, reducedMotion ? 4.8 : 2.8, delta),
        damp(currentLookAt.y, targetLookAt.y, reducedMotion ? 4.8 : 2.8, delta),
        damp(currentLookAt.z, targetLookAt.z, reducedMotion ? 4.8 : 2.8, delta),
      );

      camera.setFocalLength(42 + Math.sin(time * 0.05) * 1.2 + progress * 5);
      camera.lookAt(currentLookAt);

      const frame: ExperienceFrame = {
        progress,
        time: reducedMotion ? time * 0.25 : time,
        delta,
        pointer,
        pointerVelocity,
      };

      world.update(frame);
      particles.update(frame);
      controllers.forEach((controller) => controller.update(frame));

      renderer.render(scene, camera);
      animationFrame = window.requestAnimationFrame(render);
    };

    window.addEventListener("resize", resize);
    host.addEventListener("pointermove", move, { passive: true });
    host.addEventListener("pointerleave", resetPointer);
    resize();
    render();

    return () => {
      window.cancelAnimationFrame(animationFrame);
      window.removeEventListener("resize", resize);
      host.removeEventListener("pointermove", move);
      host.removeEventListener("pointerleave", resetPointer);
      controllers.forEach((controller) => controller.dispose());
      particles.dispose();
      world.dispose();
      renderer.dispose();
      renderer.domElement.remove();
    };
  }, [progressRef, reducedMotion]);

  return <div ref={hostRef} className="experience-canvas" aria-hidden="true" />;
}

function useMousePosition(): MousePosition {
  const [mouse, setMouse] = useState<MousePosition>({
    x: 0,
    y: 0,
    normalizedX: 0,
    normalizedY: 0,
  });

  useEffect(() => {
    const handleMove = (e: MouseEvent) => {
      setMouse({
        x: e.clientX,
        y: e.clientY,
        normalizedX: (e.clientX / window.innerWidth) * 2 - 1,
        normalizedY: -((e.clientY / window.innerHeight) * 2 - 1),
      });
    };
    window.addEventListener("mousemove", handleMove, { passive: true });
    return () => window.removeEventListener("mousemove", handleMove);
  }, []);

  return mouse;
}

function useScrollProgress(): number {
  const [progress, setProgress] = useState(0);

  useEffect(() => {
    const handleScroll = () => {
      const max =
        document.documentElement.scrollHeight - window.innerHeight;
      setProgress(max > 0 ? window.scrollY / max : 0);
    };
    window.addEventListener("scroll", handleScroll, { passive: true });
    return () => window.removeEventListener("scroll", handleScroll);
  }, []);

  return progress;
}

function useIntersection(
  ref: React.RefObject<Element | null>,
  threshold = 0.15
): boolean {
  const [visible, setVisible] = useState(false);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const obs = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setVisible(true);
          obs.disconnect();
        }
      },
      { threshold }
    );
    obs.observe(el);
    return () => obs.disconnect();
  }, [ref, threshold]);

  return visible;
}

/** Single contract address row */
function ContractRow({ contract, index }: { contract: Contract; index: number }) {
  const ref = useRef<HTMLDivElement>(null);
  const visible = useIntersection(ref as React.RefObject<Element>);
  const [copied, setCopied] = useState(false);

  const handleCopy = useCallback(async () => {
    try {
      await navigator.clipboard.writeText(contract.address);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch {
      // clipboard not available
    }
  }, [contract.address]);

  const short =
    contract.address.slice(0, 6) + "···" + contract.address.slice(-4);

  return (
    <div
      ref={ref}
      className="contract-row"
      style={{
        transitionDelay: `${index * 80}ms`,
        opacity: visible ? 1 : 0,
        transform: visible ? "translateY(0)" : "translateY(24px)",
      }}
      onClick={handleCopy}
      role="button"
      tabIndex={0}
      aria-label={`Copy ${contract.label} address`}
      onKeyDown={(e) => e.key === "Enter" && handleCopy()}
    >
      <div className="contract-row-left">
        <span className="contract-index">0{index + 1}</span>
        <div className="contract-info">
          <span className="contract-label">{contract.label}</span>
          <span className="contract-desc">{contract.description}</span>
        </div>
      </div>
      <div className="contract-row-right">
        <span className="contract-address">{short}</span>
        <div className={`copy-btn ${copied ? "copied" : ""}`}>
          {copied ? (
            <svg width="14" height="14" viewBox="0 0 24 24" fill="none">
              <polyline points="20 6 9 17 4 12" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
            </svg>
          ) : (
            <svg width="14" height="14" viewBox="0 0 24 24" fill="none">
              <rect x="9" y="9" width="13" height="13" rx="2" stroke="currentColor" strokeWidth="1.5" />
              <path d="M5 15H4a2 2 0 01-2-2V4a2 2 0 012-2h9a2 2 0 012 2v1" stroke="currentColor" strokeWidth="1.5" />
            </svg>
          )}
        </div>
      </div>
    </div>
  );
}

/** Navigation */
function Nav({ progress }: { progress: number }) {
  const [scrolled, setScrolled] = useState(false);

  useEffect(() => {
    const handler = () => setScrolled(window.scrollY > 60);
    window.addEventListener("scroll", handler, { passive: true });
    return () => window.removeEventListener("scroll", handler);
  }, []);

  return (
    <nav className={`nav ${scrolled ? "nav-scrolled" : ""}`}>
      <div className="nav-inner">
        <a href="#" className="nav-logo" aria-label="Nexus RWA">
          <span className="nav-logo-mark">N</span>
          <span className="nav-logo-text">NEXUS RWA</span>
        </a>
        <div className="nav-links">
          <a href="#protocol" className="nav-link">Protocol</a>
          <a href="#contracts" className="nav-link">Contracts</a>
          <a
            href="https://basescan.org/address/0x88bb8025dc10Cc642d2F0D10F4335EcDBdC9A594"
            target="_blank"
            rel="noopener noreferrer"
            className="nav-cta"
          >
            View on Basescan
          </a>
        </div>
      </div>
      <ProgressBar progress={progress} />
    </nav>
  );
}

import * as THREE from "three";
import { createParticleMaterial, type ParticleMaterial } from "../shaders/Materials";
import type { ExperienceFrame } from "../world/Environment";

export interface ParticleEngine {
  group: THREE.Group;
  update: (frame: ExperienceFrame) => void;
  dispose: () => void;
}

type Layer = {
  points: THREE.Points<THREE.BufferGeometry, ParticleMaterial>;
  material: ParticleMaterial;
  drift: number;
};

function createLayer(
  count: number,
  radius: number,
  height: number,
  color: string,
  scale: number,
  opacity: number,
): Layer {
  const positions = new Float32Array(count * 3);
  const seeds = new Float32Array(count);
  const scales = new Float32Array(count);

  for (let index = 0; index < count; index += 1) {
    const ratio = index / count;
    const angle = ratio * Math.PI * 27.4 + Math.sin(index * 1.71) * 0.9;
    const radial = radius * (0.22 + ((index * 37) % 100) / 100);

    positions.set(
      [
        Math.cos(angle) * radial + Math.sin(index * 0.31) * 0.3,
        ((((index * 53) % 100) / 100) - 0.5) * height + Math.cos(angle * 1.3) * 0.38,
        Math.sin(angle) * radial * 0.72 + Math.sin(index * 0.57) * 0.34,
      ],
      index * 3,
    );
    seeds[index] = ((index * 71) % 997) / 997;
    scales[index] = 0.16 + ((index * 29) % 100) / 100;
  }

  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute("position", new THREE.BufferAttribute(positions, 3));
  geometry.setAttribute("aSeed", new THREE.BufferAttribute(seeds, 1));
  geometry.setAttribute("aScale", new THREE.BufferAttribute(scales, 1));

  const material = createParticleMaterial(color);
  material.uniforms.uScale.value = scale;
  material.uniforms.uOpacity.value = opacity;

  return {
    points: new THREE.Points(geometry, material),
    material,
    drift: 0.012 + scale * 0.007,
  };
}

export function createParticleEngine(scene: THREE.Scene): ParticleEngine {
  const group = new THREE.Group();
  const layers = [
    createLayer(1150, 7.4, 8.2, "#cceff0", 0.72, 0.28),
    createLayer(860, 4.8, 5.2, "#a7dfe7", 1.03, 0.4),
    createLayer(380, 2.85, 3.25, "#dffafa", 1.38, 0.5),
  ];

  layers.forEach((layer) => group.add(layer.points));
  scene.add(group);

  return {
    group,

    update(frame) {
      const ordered = Math.min(Math.max((frame.progress - 0.35) / 0.38, 0), 1);
      const calm = Math.min(Math.max((frame.progress - 0.78) / 0.22, 0), 1);

      group.rotation.y = frame.time * 0.014 + frame.pointer.x * 0.065;
      group.rotation.x = frame.pointer.y * 0.035;
      group.position.y = Math.sin(frame.time * 0.09) * 0.08;
      group.scale.setScalar(1 - calm * 0.08);

      layers.forEach((layer, index) => {
        layer.points.rotation.z = frame.time * layer.drift * (index % 2 === 0 ? 1 : -1);
        layer.material.uniforms.uTime.value = frame.time + index * 7.3;
        layer.material.uniforms.uProgress.value = ordered;
        layer.material.uniforms.uPointer.value.copy(frame.pointer);
        layer.material.uniforms.uOpacity.value =
          (index === 0 ? 0.28 : index === 1 ? 0.4 : 0.52) * (0.72 + ordered * 0.34 - calm * 0.12);
      });
    },

    dispose() {
      layers.forEach((layer) => {
        layer.points.geometry.dispose();
        layer.material.dispose();
      });
      scene.remove(group);
    },
  };
}

import { createFilamentMaterial, createParticleMaterial } from "../shaders/Materials";
import type { ExperienceFrame, ExperienceWorld, SceneController } from "../world/Environment";

const clamp = (value: number) => Math.min(Math.max(value, 0), 1);
const smooth = (value: number) => value * value * (3 - 2 * value);

export function createFeatureScene(world: ExperienceWorld): SceneController {
  const group = new THREE.Group();
  const lineMaterial = createFilamentMaterial("#78d8e4");
  const pointMaterial = createParticleMaterial("#e4ffff");
  const trajectories: THREE.Line[] = [];
  const packets: THREE.Points[] = [];

  pointMaterial.uniforms.uScale.value = 1.55;
  pointMaterial.uniforms.uOpacity.value = 0;

  for (let index = 0; index < 17; index += 1) {
    const phase = (index / 17) * Math.PI * 2;
    const curve = new THREE.CatmullRomCurve3([
      new THREE.Vector3(Math.cos(phase) * 5.7, Math.sin(phase * 1.9) * 2.6, -2.5),
      new THREE.Vector3(Math.cos(phase + 0.68) * 2.65, Math.sin(phase + 0.72) * 1.6, -0.45),
      new THREE.Vector3(Math.cos(phase + 1.35) * 1.1, Math.sin(phase + 1.4) * 0.72, 0.55),
      new THREE.Vector3(Math.cos(phase + 2.36) * 5.9, Math.sin(phase + 2.1) * 2.7, 2.65),
    ]);

    const line = new THREE.Line(
      new THREE.BufferGeometry().setFromPoints(curve.getPoints(132)),
      lineMaterial,
    );
    group.add(line);
    trajectories.push(line);

    const packetCount = 5;
    const positions = new Float32Array(packetCount * 3);
    const seeds = new Float32Array(packetCount);
    const scales = new Float32Array(packetCount);

    for (let pointIndex = 0; pointIndex < packetCount; pointIndex += 1) {
      const point = curve.getPointAt((pointIndex + 0.5) / packetCount);
      positions.set([point.x, point.y, point.z], pointIndex * 3);
      seeds[pointIndex] = (index * 17 + pointIndex * 31) / 331;
      scales[pointIndex] = 0.5 + pointIndex * 0.1;
    }

    const packetGeometry = new THREE.BufferGeometry();
    packetGeometry.setAttribute("position", new THREE.BufferAttribute(positions, 3));
    packetGeometry.setAttribute("aSeed", new THREE.BufferAttribute(seeds, 1));
    packetGeometry.setAttribute("aScale", new THREE.BufferAttribute(scales, 1));

    const packet = new THREE.Points(packetGeometry, pointMaterial);
    group.add(packet);
    packets.push(packet);
  }

  group.rotation.x = -0.05;
  world.root.add(group);

  return {
    update(frame: ExperienceFrame) {
      const entrance = smooth(clamp((frame.progress - 0.3) / 0.18));
      const exit = smooth(clamp((frame.progress - 0.73) / 0.22));
      const presence = entrance * (1 - exit);

      group.visible = presence > 0.004;
      group.rotation.y = frame.time * 0.024 + frame.progress * 0.55;
      group.rotation.z = Math.sin(frame.time * 0.1) * 0.025;
      group.scale.setScalar(0.82 + presence * 0.27);

      lineMaterial.uniforms.uTime.value = frame.time;
      lineMaterial.uniforms.uOpacity.value = presence * 0.3;
      pointMaterial.uniforms.uTime.value = frame.time;
      pointMaterial.uniforms.uProgress.value = entrance;
      pointMaterial.uniforms.uPointer.value.copy(frame.pointer);
      pointMaterial.uniforms.uOpacity.value = presence * 0.82;

      trajectories.forEach((trajectory, index) => {
        trajectory.rotation.x = Math.sin(frame.time * 0.08 + index * 0.37) * 0.1;
        trajectory.rotation.y = Math.cos(frame.time * 0.06 + index * 0.51) * 0.1;
      });

      packets.forEach((packet, index) => {
        packet.rotation.y = frame.time * (0.025 + index * 0.001);
      });
    },

    dispose() {
      trajectories.forEach((trajectory) => trajectory.geometry.dispose());
      packets.forEach((packet) => packet.geometry.dispose());
      lineMaterial.dispose();
      pointMaterial.dispose();
      world.root.remove(group);
    },
  };
}

"use client";

import { useEffect, useRef, useState } from "react";

export interface ScrollProgressState {
  progress: number;
  progressRef: React.MutableRefObject<number>;
  reducedMotion: boolean;
}

const clamp = (value: number) => Math.min(Math.max(value, 0), 1);

export function useScrollProgress(): ScrollProgressState {
  const progressRef = useRef(0);
  const [progress, setProgress] = useState(0);
  const [reducedMotion, setReducedMotion] = useState(false);

  useEffect(() => {
    const motionQuery = window.matchMedia("(prefers-reduced-motion: reduce)");
    let animationFrame = 0;

    const updateProgress = () => {
      const maximumScroll = Math.max(
        document.documentElement.scrollHeight - window.innerHeight,
        1,
      );
      const nextProgress = clamp(window.scrollY / maximumScroll);

      progressRef.current = nextProgress;
      setProgress((currentProgress) =>
        Math.abs(currentProgress - nextProgress) > 0.0005
          ? nextProgress
          : currentProgress,
      );
    };

    const requestUpdate = () => {
      window.cancelAnimationFrame(animationFrame);
      animationFrame = window.requestAnimationFrame(updateProgress);
    };

    const updateMotionPreference = () => {
      setReducedMotion(motionQuery.matches);
    };

    updateMotionPreference();
    updateProgress();

    window.addEventListener("scroll", requestUpdate, { passive: true });
    window.addEventListener("resize", requestUpdate, { passive: true });
    motionQuery.addEventListener("change", updateMotionPreference);

    return () => {
      window.cancelAnimationFrame(animationFrame);
      window.removeEventListener("scroll", requestUpdate);
      window.removeEventListener("resize", requestUpdate);
      motionQuery.removeEventListener("change", updateMotionPreference);
    };
  }, []);

  return { progress, progressRef, reducedMotion };
}

  delta: number;

export interface SceneController {

  root: THREE.Group;
  field: THREE.Group;

    aegis: AegisMaterial;
    filaments: FilamentMaterial;

type Lamella = {
  mesh: THREE.Mesh<THREE.BufferGeometry, AegisMaterial>;
  phase: number;
  height: number;
};

const clamp = (value: number) => Math.min(Math.max(value, 0), 1);
const smooth = (value: number) => value * value * (3 - 2 * value);

export function createRibbonGeometry(
  radius: number,
  height: number,
  phase: number,
  turns = 1.25,
  segments = 92,
): THREE.BufferGeometry {
  const vertices: number[] = [];
  const phases: number[] = [];
  const lifts: number[] = [];
  const indices: number[] = [];

  for (let index = 0; index <= segments; index += 1) {
    const progress = index / segments;
    const angle = phase + progress * Math.PI * 2 * turns;
    const undulation =
      Math.sin(progress * Math.PI * 2 + phase * 1.7) * 0.24 +
      Math.sin(progress * Math.PI * 5.2 + phase * 0.63) * 0.08;
    const center = new THREE.Vector3(
      Math.cos(angle) * (radius + undulation),
      (progress - 0.5) * height + Math.sin(angle * 1.35) * 0.34,
      Math.sin(angle) * (radius + undulation) * 0.74,
    );
    const outward = new THREE.Vector3(Math.cos(angle), 0.12, Math.sin(angle) * 0.74).normalize();
    const width = 0.09 + Math.sin(progress * Math.PI) * 0.075;
    const left = center.clone().addScaledVector(outward, -width);
    const right = center.clone().addScaledVector(outward, width);

    vertices.push(left.x, left.y, left.z, right.x, right.y, right.z);
    phases.push(phase / (Math.PI * 2), phase / (Math.PI * 2));
    lifts.push(progress, progress);

    if (index < segments) {
      const offset = index * 2;
      indices.push(offset, offset + 1, offset + 2, offset + 1, offset + 3, offset + 2);
    }
  }

  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute("position", new THREE.Float32BufferAttribute(vertices, 3));
  geometry.setAttribute("aPhase", new THREE.Float32BufferAttribute(phases, 1));
  geometry.setAttribute("aLift", new THREE.Float32BufferAttribute(lifts, 1));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  return geometry;
}

function createField(material: AegisMaterial) {
  const group = new THREE.Group();
  const lamellae: Lamella[] = [];

  for (let index = 0; index < 31; index += 1) {
    const phase = (index / 31) * Math.PI * 2 + Math.sin(index * 1.73) * 0.17;
    const height = 2.1 + (index % 6) * 0.39;
    const mesh = new THREE.Mesh(
      createRibbonGeometry(1.4 + (index % 8) * 0.19, height, phase),
      material,
    );

    mesh.rotation.set(
      Math.sin(index * 0.91) * 0.46,
      Math.cos(index * 0.57) * 0.44,
      Math.sin(index * 0.37) * 0.3,
    );

    group.add(mesh);
    lamellae.push({ mesh, phase, height });
  }

  return { group, lamellae };
}

function createFilamentField(material: FilamentMaterial) {
  const group = new THREE.Group();

  for (let index = 0; index < 16; index += 1) {
    const phase = (index / 16) * Math.PI * 2;
    const curve = new THREE.CatmullRomCurve3([
      new THREE.Vector3(Math.cos(phase) * 4.4, Math.sin(phase * 1.8) * 1.9, -2.1),
      new THREE.Vector3(Math.cos(phase + 0.76) * 2.4, Math.sin(phase + 0.4) * 1.15, -0.55),
      new THREE.Vector3(Math.cos(phase + 1.4) * 1.2, Math.sin(phase + 1.2) * 0.85, 0.7),
      new THREE.Vector3(Math.cos(phase + 2.16) * 4.8, Math.sin(phase + 1.8) * 2.1, 2.15),
    ]);

    group.add(
      new THREE.Line(
        new THREE.BufferGeometry().setFromPoints(curve.getPoints(100)),
        material,
      ),
    );
  }

  return group;
}

export function createEnvironment(scene: THREE.Scene): ExperienceWorld {
  scene.background = new THREE.Color("#edf3f2");
  scene.fog = new THREE.FogExp2("#edf5f4", 0.036);

  const root = new THREE.Group();
  const materials = {
    aegis: createAegisMaterial(),
    filaments: createFilamentMaterial(),
  };
  const { group: field, lamellae } = createField(materials.aegis);

  root.add(field, createFilamentField(materials.filaments));
  scene.add(root);

  const hemisphere = new THREE.HemisphereLight("#f9ffff", "#abc6ce", 2.15);
  const key = new THREE.DirectionalLight("#eaffff", 2.2);
  const rim = new THREE.PointLight("#9ce8f2", 14, 18, 2);
  const depth = new THREE.PointLight("#d6e2ff", 7, 20, 2);

  key.position.set(-4.8, 6.6, 4.2);
  rim.position.set(4.2, 0.8, 4.4);
  depth.position.set(-2.8, -3.8, -5.2);
  scene.add(hemisphere, key, rim, depth);

  const update = (frame: ExperienceFrame) => {
    const entrance = smooth(clamp((frame.progress - 0.06) / 0.22));
    const intelligence = smooth(clamp((frame.progress - 0.34) / 0.3));
    const resolve = smooth(clamp((frame.progress - 0.61) / 0.23));
    const finality = smooth(clamp((frame.progress - 0.8) / 0.2));

    root.rotation.y = frame.time * 0.026 + frame.pointer.x * 0.075;
    root.rotation.x = Math.sin(frame.time * 0.07) * 0.035 - frame.pointer.y * 0.035;
    root.position.y = Math.sin(frame.time * 0.12) * 0.11 - finality * 0.24;
    root.scale.setScalar(0.74 + entrance * 0.26 + intelligence * 0.05);

    field.rotation.y = -frame.time * 0.038 + intelligence * 0.22;
    field.rotation.z = Math.sin(frame.time * 0.09) * 0.045 - finality * 0.055;
    field.position.z = -0.65 + entrance * 0.5 + finality * 0.68;

    lamellae.forEach(({ mesh, phase, height }, index) => {
      const activity = Math.sin(frame.time * (0.11 + (index % 5) * 0.012) + phase * 5.4) * 0.5 + 0.5;
      const precision = intelligence * (0.32 + activity * 0.68);

      mesh.position.set(
        Math.sin(phase * 2.3 + frame.time * 0.06) * (0.16 + intelligence * 0.11),
        Math.cos(phase * 1.8 + frame.time * 0.05) * (0.14 + resolve * 0.08) + (height - 2.1) * 0.012,
        Math.cos(phase * 1.7) * 0.14,
      );
      mesh.rotation.y = phase * 0.23 + precision * 0.32;
      mesh.rotation.x = Math.sin(phase * 1.73 + frame.time * 0.05) * 0.22 - resolve * 0.13;
      mesh.scale.set(1 + precision * 0.08, 1 + resolve * 0.14, 1 + precision * 0.12);
    });

    materials.aegis.uniforms.uTime.value = frame.time;
    materials.aegis.uniforms.uProgress.value = frame.progress;
    materials.aegis.uniforms.uEnergy.value = 0.1 + intelligence * 0.55 + resolve * 0.16;
    materials.filaments.uniforms.uTime.value = frame.time;
    materials.filaments.uniforms.uOpacity.value = 0.05 + entrance * 0.16 + intelligence * 0.22 - finality * 0.12;

    key.intensity = 1.2 + entrance * 1.2 + finality * 0.45;
    rim.intensity = 5 + intelligence * 16 + resolve * 10;
    depth.intensity = 3.5 + resolve * 8;
    rim.position.x = 3.5 + Math.sin(frame.time * 0.13) * 1.2;

    if (scene.fog instanceof THREE.FogExp2) {
      scene.fog.density = 0.044 - entrance * 0.015 - intelligence * 0.008 + finality * 0.004;
      scene.fog.color.set(finality > 0.35 ? "#f5f8f7" : "#edf5f4");
    }

    if (scene.background instanceof THREE.Color) {
      scene.background.set(finality > 0.35 ? "#f8fbfa" : "#edf3f2");
    }
  };

  const dispose = () => {
    root.traverse((object) => {
      if (object instanceof THREE.Mesh || object instanceof THREE.Line || object instanceof THREE.Points) {
        object.geometry.dispose();
      }
    });

    materials.aegis.dispose();
    materials.filaments.dispose();
    scene.remove(root, hemisphere, key, rim, depth);
  };

  return { root, field, materials, update, dispose };
}

import * as THREE from "three";

export type AegisMaterial = THREE.ShaderMaterial & {

    uTime: { value: number };
    uProgress: { value: number };
    uEnergy: { value: number };
    uBaseColor: { value: THREE.Color };
    uEdgeColor: { value: THREE.Color };
  };
};

export type FilamentMaterial = THREE.ShaderMaterial & {
  uniforms: {
    uTime: { value: number };
    uOpacity: { value: number };
    uColor: { value: THREE.Color };
  };
};

export type ParticleMaterial = THREE.ShaderMaterial & {
  uniforms: {
    uTime: { value: number };
    uProgress: { value: number };
    uOpacity: { value: number };
    uScale: { value: number };
    uColor: { value: THREE.Color };
    uPointer: { value: THREE.Vector2 };
  };
};

export function createAegisMaterial(
  baseColor = "#dceef1",
  edgeColor = "#7fdbe5",
): AegisMaterial {
  return new THREE.ShaderMaterial({
    transparent: true,
    depthWrite: true,
    side: THREE.DoubleSide,
    uniforms: {
      uTime: { value: 0 },
      uProgress: { value: 0 },
      uEnergy: { value: 0.1 },
      uBaseColor: { value: new THREE.Color(baseColor) },
      uEdgeColor: { value: new THREE.Color(edgeColor) },
    },
    vertexShader: `
      uniform float uTime;
      uniform float uProgress;
      uniform float uEnergy;
      attribute float aPhase;
      attribute float aLift;
      varying vec3 vNormal;
      varying vec3 vViewPosition;
      varying float vPhase;
      varying float vLift;

      float wave(float value) {
        return sin(value) * 0.52 + sin(value * 0.47 + 1.7) * 0.27 + sin(value * 1.91 + 0.8) * 0.12;
      }

      void main() {
        float motion = wave(aPhase * 8.0 + uTime * 0.22 + uProgress * 4.0);
        vec3 displaced = position + normal * motion * mix(0.018, 0.105, uEnergy);
        displaced += normal * sin(aLift * 9.0 + uTime * 0.16) * 0.018;
        vec4 modelPosition = modelMatrix * vec4(displaced, 1.0);
        vec4 modelViewPosition = viewMatrix * modelPosition;
        vNormal = normalize(normalMatrix * normal);
        vViewPosition = -modelViewPosition.xyz;
        vPhase = aPhase;
        vLift = aLift;
        gl_Position = projectionMatrix * modelViewPosition;
      }
    `,
    fragmentShader: `
      uniform float uTime;
      uniform float uProgress;
      uniform float uEnergy;
      uniform vec3 uBaseColor;
      uniform vec3 uEdgeColor;
      varying vec3 vNormal;
      varying vec3 vViewPosition;
      varying float vPhase;
      varying float vLift;

      void main() {
        float fresnel = pow(1.0 - max(dot(normalize(vNormal), normalize(vViewPosition)), 0.0), 2.35);
        float striation = sin(vLift * 72.0 + vPhase * 17.0 + uTime * 0.12) * 0.5 + 0.5;
        float flow = sin(vPhase * 11.0 - uTime * 0.16 + uProgress * 5.5) * 0.5 + 0.5;
        vec3 color = mix(uBaseColor * 0.72, uEdgeColor, fresnel * 0.84);
        color += uEdgeColor * striation * 0.075;
        color += vec3(0.84, 0.96, 1.0) * flow * uEnergy * 0.18;
        float alpha = (mix(0.19, 0.72, fresnel) + striation * 0.08) * mix(0.82, 1.0, flow);
        gl_FragColor = vec4(color, alpha);
      }
    `,
  }) as AegisMaterial;
}

export function createFilamentMaterial(color = "#9adfe8"): FilamentMaterial {
  return new THREE.ShaderMaterial({
    transparent: true,
    depthWrite: false,
    blending: THREE.AdditiveBlending,
    uniforms: {
      uTime: { value: 0 },
      uOpacity: { value: 0 },
      uColor: { value: new THREE.Color(color) },
    },
    vertexShader: `
      uniform float uTime;
      varying float vPulse;

      void main() {
        vec4 modelPosition = modelMatrix * vec4(position, 1.0);
        vec4 modelViewPosition = viewMatrix * modelPosition;
        vPulse = sin(position.x * 1.9 + position.y * 2.7 + position.z * 1.3 + uTime * 0.32) * 0.5 + 0.5;
        gl_Position = projectionMatrix * modelViewPosition;
      }
    `,
    fragmentShader: `
      uniform float uOpacity;
      uniform vec3 uColor;
      varying float vPulse;

      void main() {
        gl_FragColor = vec4(uColor, uOpacity * mix(0.36, 1.0, vPulse));
      }
    `,
  }) as FilamentMaterial;
}

export function createParticleMaterial(color = "#b5f2f4"): ParticleMaterial {
  return new THREE.ShaderMaterial({
    transparent: true,
    depthWrite: false,
    blending: THREE.AdditiveBlending,
    uniforms: {
      uTime: { value: 0 },
      uProgress: { value: 0 },
      uOpacity: { value: 0.6 },
      uScale: { value: 1 },
      uColor: { value: new THREE.Color(color) },
      uPointer: { value: new THREE.Vector2() },
    },
    vertexShader: `
      uniform float uTime;
      uniform float uProgress;
      uniform float uScale;
      uniform vec2 uPointer;
      attribute float aSeed;
      attribute float aScale;
      varying float vAlpha;

      void main() {
        vec3 transformed = position;
        float phase = aSeed * 6.28318530718;
        transformed.x += sin(phase + uTime * 0.09) * 0.11;
        transformed.y += sin(uTime * (0.08 + aSeed * 0.12) + phase) * 0.16;
        transformed.z += cos(phase * 1.7 + uTime * 0.07) * 0.09;
        vec2 field = transformed.xy - uPointer * 2.2;
        transformed.xy += normalize(field + 0.0001) * smoothstep(1.65, 0.12, length(field)) * 0.035;
        vec4 modelViewPosition = modelViewMatrix * vec4(transformed, 1.0);
        gl_PointSize = (1.4 + aScale * 2.3) * uScale * clamp(24.0 / max(1.0, -modelViewPosition.z), 0.55, 2.2);
        gl_Position = projectionMatrix * modelViewPosition;
        vAlpha = mix(0.34, 1.0, aScale) * mix(0.86, 1.0, uProgress);
      }
    `,
    fragmentShader: `
      uniform float uOpacity;
      uniform vec3 uColor;
      varying float vAlpha;

      void main() {
        float alpha = smoothstep(0.5, 0.03, length(gl_PointCoord - 0.5)) * uOpacity * vAlpha;
        gl_FragColor = vec4(uColor, alpha);
      }
    `,
  }) as ParticleMaterial;
}

export const clamp = (value: number, minimum = 0, maximum = 1) =>
  Math.min(Math.max(value, minimum), maximum);

export const lerp = (from: number, to: number, amount: number) =>
  from + (to - from) * amount;

export const smoothstep = (minimum: number, maximum: number, value: number) => {
  const normalized = clamp((value - minimum) / (maximum - minimum));
  return normalized * normalized * (3 - 2 * normalized);
};

export const remap = (
  value: number,
  fromMinimum: number,
  fromMaximum: number,
  toMinimum: number,
  toMaximum: number,
) => toMinimum + ((value - fromMinimum) / (fromMaximum - fromMinimum)) * (toMaximum - toMinimum);

export const damp = (current: number, target: number, smoothing: number, delta: number) =>
  lerp(current, target, 1 - Math.exp(-smoothing * delta));

export const easeInOutCubic = (value: number) =>
  value < 0.5 ? 4 * value * value * value : 1 - Math.pow(-2 * value + 2, 3) / 2;

export const easeOutExpo = (value: number) =>
  value >= 1 ? 1 : 1 - Math.pow(2, -10 * value);

export const degToRad = (degrees: number) => (degrees * Math.PI) / 180;

export const radToDeg = (radians: number) => (radians * 180) / Math.PI;
